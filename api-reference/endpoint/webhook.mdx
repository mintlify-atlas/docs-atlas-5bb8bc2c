## Overview

The Plant Store API sends webhook notifications to your configured endpoint when new plants are added. This allows your application to react in real-time to inventory changes without constant polling.

<Note>
Webhooks are POST requests sent from the Plant Store API to your application's endpoint. You must provide a publicly accessible HTTPS URL to receive webhook events.
</Note>

## Webhook Configuration

To receive webhook events, you need to:

1. Set up a publicly accessible HTTPS endpoint
2. Register your webhook URL in the Plant Store dashboard
3. Implement the webhook handler to process incoming events
4. Return a `200` status code to acknowledge receipt

<Warning>
Your webhook endpoint must respond with a 200 status code within 5 seconds. Failed webhooks will be retried up to 3 times with exponential backoff.
</Warning>

## Request Format

### Headers

<ParamField header="Content-Type" type="string">
  Always set to `application/json`
</ParamField>

<ParamField header="X-Webhook-Signature" type="string">
  HMAC-SHA256 signature of the request body using your webhook secret. Use this to verify the webhook authenticity.
</ParamField>

<ParamField header="X-Webhook-ID" type="string">
  Unique identifier for this webhook event. Use for deduplication.
</ParamField>

<ParamField header="X-Webhook-Timestamp" type="string">
  Unix timestamp when the event was generated
</ParamField>

### Payload

<ResponseField name="event" type="string">
  The event type. For new plants, this is `plant.created`
</ResponseField>

<ResponseField name="timestamp" type="string">
  ISO 8601 timestamp when the event occurred
</ResponseField>

<ResponseField name="data" type="object">
  The plant data object
  
  <Expandable title="Plant Data">
    <ResponseField name="id" type="integer">
      Unique identifier for the plant
    </ResponseField>
    
    <ResponseField name="name" type="string">
      The name of the plant
    </ResponseField>
    
    <ResponseField name="tag" type="string">
      Optional tag categorizing the plant type
    </ResponseField>
  </Expandable>
</ResponseField>

```go Go
package main

import (
    "crypto/hmac"
    "crypto/sha256"
    "encoding/hex"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "os"
)

type WebhookPayload struct {
    Event     string      `json:"event"`
    Timestamp string      `json:"timestamp"`
    Data      PlantData   `json:"data"`
}

type PlantData struct {
    ID   int    `json:"id"`
    Name string `json:"name"`
    Tag  string `json:"tag,omitempty"`
}

func verifySignature(payload []byte, signature string) bool {
    secret := os.Getenv("WEBHOOK_SECRET")
    mac := hmac.New(sha256.New, []byte(secret))
    mac.Write(payload)
    expectedSignature := hex.EncodeToString(mac.Sum(nil))
    return hmac.Equal([]byte(signature), []byte(expectedSignature))
}

func plantWebhookHandler(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
        return
    }
    
    signature := r.Header.Get("X-Webhook-Signature")
    
    body, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "Error reading body", http.StatusBadRequest)
        return
    }
    defer r.Body.Close()
    
    // Verify signature
    if !verifySignature(body, signature) {
        http.Error(w, "Invalid signature", http.StatusUnauthorized)
        return
    }
    
    // Parse payload
    var payload WebhookPayload
    if err := json.Unmarshal(body, &payload); err != nil {
        http.Error(w, "Invalid JSON", http.StatusBadRequest)
        return
    }
    
    // Process event
    if payload.Event == "plant.created" {
        fmt.Printf("New plant added: %s (ID: %d)\n", 
            payload.Data.Name, payload.Data.ID)
        
        // Your business logic here
    }
    
    // Acknowledge receipt
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(map[string]bool{"received": true})
}

func main() {
    http.HandleFunc("/plant/webhook", plantWebhookHandler)
    fmt.Println("Webhook server listening on :3000")
    http.ListenAndServe(":3000", nil)
}
```

## Security Best Practices

<AccordionGroup>
  <Accordion title="Always verify webhook signatures">
    Never process webhook events without verifying the signature. This prevents malicious actors from sending fake webhooks to your endpoint.
    
    ```javascript
    function verifyWebhookSignature(payload, signature, secret) {
      const hmac = crypto.createHmac('sha256', secret);
      const digest = hmac.update(JSON.stringify(payload)).digest('hex');
      
      return crypto.timingSafeEqual(
        Buffer.from(signature),
        Buffer.from(digest)
      );
    }
    ```
  </Accordion>
  
  <Accordion title="Implement replay attack prevention">
    Use the webhook ID and timestamp to prevent replay attacks by checking if you've already processed an event.
    
    ```javascript
    const processedWebhooks = new Set();
    
    function processWebhook(webhookId, timestamp, payload) {
      // Check if already processed
      if (processedWebhooks.has(webhookId)) {
        console.log('Duplicate webhook, skipping');
        return;
      }
      
      // Check timestamp is recent (within 5 minutes)
      const age = Date.now() - (timestamp * 1000);
      if (age > 300000) {
        console.log('Webhook too old, rejecting');
        return;
      }
      
      // Process the webhook
      // ...
      
      // Mark as processed
      processedWebhooks.add(webhookId);
    }
    ```
  </Accordion>
  
  <Accordion title="Use HTTPS only">
    Always configure your webhook endpoint to use HTTPS. Never accept webhooks over plain HTTP as this exposes your webhook secret and data to potential interception.
  </Accordion>
  
  <Accordion title="Store webhook secrets securely">
    Never hardcode webhook secrets in your code. Use environment variables or secure secret management services.
    
    ```javascript
    // Good - use environment variables
    const WEBHOOK_SECRET = process.env.WEBHOOK_SECRET;
    
    // Bad - hardcoded secret
    const WEBHOOK_SECRET = 'my-secret-key'; // Don't do this!
    ```
  </Accordion>
</AccordionGroup>

## Error Handling

<Warning>
If your webhook endpoint returns a non-200 status code or times out, the Plant Store API will retry the webhook delivery up to 3 times with exponential backoff (1 minute, 5 minutes, 15 minutes).
</Warning>

Handle errors gracefully and return appropriate status codes:

```javascript
app.post('/plant/webhook', async (req, res) => {
  try {
    // Verify signature
    const signature = req.headers['x-webhook-signature'];
    if (!verifyWebhookSignature(req.body, signature)) {
      return res.status(401).json({ error: 'Invalid signature' });
    }
    
    // Process webhook
    await processPlantEvent(req.body);
    
    // Success
    res.status(200).json({ received: true });
  } catch (error) {
    console.error('Webhook processing error:', error);
    
    // Return 200 anyway to prevent retries for errors we can't fix
    // Log the error for manual investigation
    res.status(200).json({ 
      received: true, 
      note: 'Processing failed but acknowledged' 
    });
  }
});
```
